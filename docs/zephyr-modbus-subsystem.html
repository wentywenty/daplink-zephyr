01概述与协议背景

Modbus 是一种工业领域广泛使用的串行通信协议，最初由 Modicon（现施耐德电气）于 1979 年发布。 Zephyr RTOS 的 Modbus 子系统实现了 Modbus 协议的核心功能，支持以下传输模式与角色：

特性说明
RTU二进制传输，CRC16 校验，高效率，工业主流
ASCIIASCII 字符传输，LRC 校验，可读性好，调试方便
RAW原始 ADU 模式，用于构建 TCP 网关等上层协议栈
Client主站（Master），主动发起请求
Server从站（Slave），被动响应请求
许可证溯源 Zephyr Modbus 实现部分源自 Silicon Labs 的 uC/Modbus 开源栈（Apache-2.0），包括客户端 API 和串行线路处理。 文件头部均有明确标注。

支持的功能码

功能码名称源码常量
01读线圈MODBUS_FC01_COIL_RD
02读离散输入MODBUS_FC02_DI_RD
03读保持寄存器MODBUS_FC03_HOLDING_REG_RD
04读输入寄存器MODBUS_FC04_IN_REG_RD
05写单线圈MODBUS_FC05_COIL_WR

03核心数据结构

3.1 modbus_adu — 应用数据单元 modbus.h:72

modbus_adu 是 Modbus 子系统的核心帧结构，用于内部处理和 Raw ADU 支持。 它统一了 RTU、ASCII 和 TCP 三种模式的帧表示。 

struct modbus_adu {
    uint16_t trans_id;      // 事务标识符（TCP 模式使用）
    uint16_t proto_id;      // 协议标识符（TCP: 0x0000）
    uint16_t length;        // 数据长度（不含 unit_id + FC）
    uint8_t  unit_id;       // 从站地址
    uint8_t  fc;            // 功能码
    uint8_t  data[CONFIG_MODBUS_BUFFER_SIZE - 4]; // 数据域
    uint16_t crc;           // RTU CRC16 / ASCII LRC
};
MBAP 头映射 trans_id + proto_id + length + unit_id 恰好对应 Modbus TCP 的 MBAP (Modbus Application Protocol) 头部的 7 字节。 RTU 模式仅使用 unit_id、fc、data、crc 字段。

3.3 modbus_context — 接口上下文 modbus_internal.h:93

每个 Modbus 接口（由设备树定义）对应一个 modbus_context 实例， 是整个子系统的运行时核心。 

struct modbus_context {
    const char *iface_name;                // 接口名称（来自设备树）
    union {
        struct modbus_serial_config *cfg;  // 串行线路配置
        struct modbus_raw_cb rawcb;        // RAW TX 回调
    };
    enum modbus_mode mode;                 // RTU / ASCII / RAW
    bool client;                           // true = 客户端
    uint32_t rxwait_to;                    // 客户端等待响应超时
    struct modbus_user_callbacks *mbs_user_cb; // 服务器用户回调
    atomic_t state;                        // 接口状态位

    struct k_mutex iface_lock;             // 客户端互斥锁
    struct k_sem client_wait_sem;          // 客户端等待信号量
    struct k_work server_work;             // RX 处理工作项
    struct modbus_adu rx_adu;              // 接收帧
    struct modbus_adu tx_adu;              // 发送帧
    int rx_adu_err;                        // 帧接收错误码

#ifdef CONFIG_MODBUS_FC08_DIAGNOSTIC
    uint16_t mbs_msg_ctr;                  // 消息计数器
    uint16_t mbs_crc_err_ctr;              // CRC 错误计数器
    uint16_t mbs_except_ctr;               // 异常计数器
    uint16_t mbs_server_msg_ctr;           // 服务器消息计数器
    uint16_t mbs_noresp_ctr;               // 无响应计数器
#endif
    sys_slist_t user_defined_cbs;          // 自定义功能码链表
    uint8_t unit_id;                       // 服务器从站地址
};
内存开销 每个 modbus_context 包含两个 modbus_adu（rx + tx）， 默认 CONFIG_MODBUS_BUFFER_SIZE=256 时，单实例约占用 1KB+ RAM。 加上 modbus_serial_config 的 UART 缓冲区（256 字节）， 一个接口约需 ~2.5KB RAM。

5.4 禁用接口 modbus_core.c:376-409

int modbus_disable(const uint8_t iface)
{
    // 1. 关闭底层传输
    switch (ctx->mode) {
    case MODBUS_MODE_RTU:
    case MODBUS_MODE_ASCII:
        modbus_serial_disable(ctx);   // TX off, RX off, stop timer
        break;
    case MODBUS_MODE_RAW:
        break;
    }

    // 2. 取消挂起的工作项
    k_work_cancel_sync(&ctx->server_work, &work_sync);

    // 3. 清除配置状态
    atomic_clear_bit(&ctx->state, MODBUS_STATE_CONFIGURED);
}

06串行线路层 (Serial Line)

串行线路层是 Modbus 最底层的传输层，直接与 UART 硬件交互。 支持两种 UART API：中断驱动 (IRQ) 和异步 (Async)。

6.1 双 API 支持 modbus_serial.c

特性IRQ 模式 (默认)Async 模式
Kconfig 默认启用 CONFIG_MODBUS_SERIAL_ASYNC_API
注册回调 uart_irq_callback_user_data_set() uart_callback_set()
TX 发送 uart_irq_tx_enable() + FIFO 填充 uart_tx()
RX 接收 uart_irq_rx_enable() + FIFO 读取 uart_rx_enable()

6.2 RS-485 收发控制

RS-485 半双工总线需要通过 GPIO 控制 DE（驱动器使能）和 nRE（接收使能）引脚。 这两个引脚在设备树中可选配置。 

// 发送前：DE = 1（使能驱动器）
static void modbus_serial_tx_on(struct modbus_context *ctx)
{
    if (cfg->de != NULL) gpio_pin_set_dt(cfg->de, 1);
    // ... 启动 TX ...
}

// 发送完成：DE = 0, 恢复接收
static void modbus_serial_tx_off(struct modbus_context *ctx)
{
    // ... 停止 TX ...
    if (cfg->de != NULL) gpio_pin_set_dt(cfg->de, 0);
}
发送完成等待 IRQ 模式中，TX 完成判断使用 uart_irq_tx_complete()， 确保 UART 移位寄存器中所有数据已发送完毕后才切换 RS-485 方向。 过早切换会导致最后一字节被截断。 modbus_serial.c:424

UART IRQ 触发
     │

7.2 RTU 接收：CRC 校验 modbus_serial.c:284-319

static int modbus_rtu_rx_adu(struct modbus_context *ctx)
{
    // 1. 长度检查：最小 4 字节 (addr + fc + crc16)
    if ((cfg->uart_buf_ctr < MODBUS_RTU_MIN_MSG_SIZE) ||
        (cfg->uart_buf_ctr > CONFIG_MODBUS_BUFFER_SIZE))
        return -EMSGSIZE;

    // 2. 提取字段
    ctx->rx_adu.unit_id = cfg->uart_buf[0];
    ctx->rx_adu.fc      = cfg->uart_buf[1];
    ctx->rx_adu.length  = cfg->uart_buf_ctr - 4;  // 减去 addr + fc + crc

    // 3. CRC 校验
    ctx->rx_adu.crc = sys_get_le16(&cfg->uart_buf[crc_idx]);
    calc_crc = crc16_ansi(&cfg->uart_buf[0],
                          cfg->uart_buf_ctr - sizeof(ctx->rx_adu.crc));
    if (ctx->rx_adu.crc != calc_crc)
        return -EIO;

    return 0;
}

7.3 RTU 发送：CRC 附加 modbus_serial.c:321-346

static void modbus_rtu_tx_adu(struct modbus_context *ctx)
{
    cfg->uart_buf[0] = ctx->tx_adu.unit_id;
    cfg->uart_buf[1] = ctx->tx_adu.fc;
    tx_bytes = 2 + ctx->tx_adu.length;
    memcpy(data_ptr, ctx->tx_adu.data, ctx->tx_adu.length);

    // 计算 CRC 并附加到帧尾 (小端序)
    ctx->tx_adu.crc = crc16_ansi(&cfg->uart_buf[0],
                                  ctx->tx_adu.length + 2);
    sys_put_le16(ctx->tx_adu.crc,
                 &cfg->uart_buf[ctx->tx_adu.length + 2]);

    // 关闭 RX → 打开 TX（半双工切换）
    modbus_serial_rx_off(ctx);
    modbus_serial_tx_on(ctx);
}

08ASCII 帧处理：LRC 校验与编解码

ASCII 模式将每个字节编码为两个 ASCII 十六进制字符（如 0xFF → "FF"）， 使用 ':' 作为帧起始符，"\r\n" 作为帧结束符。 校验使用 LRC (Longitudinal Redundancy Check)。

8.1 ASCII 帧格式

┌─────┬──────────┬──────────┬──────────┬─────┬─────┬─────┐
│ ':' │ 地址 HH │ 功能 HH │ 数据 HH │ LRC │ CR  │ LF  │
│ 1B  │  1+1 = 2 chars  │ 2 chars  │ N×2  │ 2   │ 1   │ 1   │
└─────┴──────────┴──────────┴──────────┴─────┴─────┴─────┘

8.2 LRC 计算 modbus_serial.c:117-137

static uint8_t modbus_ascii_get_lrc(uint8_t *src, size_t length)
{
    uint8_t lrc = 0;

    while (length-- > 0) {
        hex2bin(pblock, 2, &tmp, sizeof(tmp));  // ASCII → binary
        lrc += tmp;
        pblock += 2;
    }

    lrc = ~lrc + 1;  // 取二进制补码
    return lrc;
}

8.3 ASCII 接收 modbus_serial.c:140-211

接收流程严格验证帧结构：

  1. 检查字节数为奇数（ASCII 模式特性）
  2. 验证最小长度 (MODBUS_ASCII_MIN_MSG_SIZE = 11)
  3. 检查起始符 ':' 和结束符 "\r\n"
  4. 逐字节 hex2bin 转换：地址 → 功能码 → 数据
  5. 提取帧内 LRC 并与计算值比较

8.4 ASCII 发送 modbus_serial.c:226-271

static void modbus_ascii_tx_adu(struct modbus_context *ctx)
{
    cfg->uart_buf[0] = ':';                       // 起始符
    pbuf = modbus_ascii_bin2hex(ctx->tx_adu.unit_id, pbuf);  // 地址
    pbuf = modbus_ascii_bin2hex(ctx->tx_adu.fc, pbuf);       // 功能码
    for (int i = 0; i < ctx->tx_adu.length; i++)
        pbuf = modbus_ascii_bin2hex(ctx->tx_adu.data[i], pbuf);

    lrc = modbus_ascii_get_lrc(...);
    pbuf = modbus_ascii_bin2hex(lrc, pbuf);       // LRC
    *pbuf++ = '\r';  *pbuf++ = '\n';              // 结束符

    modbus_serial_rx_off(ctx);
    modbus_serial_tx_on(ctx);
}

09核心 RX 处理与工作队列

无论是 RTU 定时器到期、ASCII 结束符检测，还是 Async 模式 RX_RDY 事件， 最终都提交到同一个 k_work（server_work）。

9.1 modbus_rx_handler modbus_core.c:68-115

static void modbus_rx_handler(struct k_work *item)
{
    ctx = CONTAINER_OF(item, struct modbus_context, server_work);

    // Step 1: 根据 mode 解码帧
    switch (ctx->mode) {
    case MODBUS_MODE_RTU:
    case MODBUS_MODE_ASCII:
        modbus_serial_rx_disable(ctx);
        ctx->rx_adu_err = modbus_serial_rx_adu(ctx);
        break;
    case MODBUS_MODE_RAW:
        ctx->rx_adu_err = modbus_raw_rx_adu(ctx);
        break;
    }

    // Step 2: 根据角色分发
    if (ctx->client == true) {
        k_sem_give(&ctx->client_wait_sem);    // 唤醒等待的客户端
    } else if (IS_ENABLED(CONFIG_MODBUS_SERVER)) {
        bool respond = modbus_server_handler(ctx);
        if (respond) {
            modbus_tx_adu(ctx);                  // 发送响应
        } else {
            modbus_serial_rx_enable(ctx);         // 丢弃帧，重新接收
        }
    }
}
双角色复用 同一个 work handler 处理客户端和服务器的响应帧。 对于客户端：通过信号量通知阻塞的调用线程； 对于服务器：直接调用 server_handler 处理请求并回复。

9.2 客户端发送-等待-接收 modbus_core.c:138-150

int modbus_tx_wait_rx_adu(struct modbus_context *ctx)
{
    k_sem_reset(&ctx->client_wait_sem);
    modbus_tx_adu(ctx);                                   // 发送请求

    if (k_sem_take(&ctx->client_wait_sem,
                   K_USEC(ctx->rxwait_to)) != 0) {        // 等待响应
        return -ETIMEDOUT;
    }

    return ctx->rx_adu_err;                               // 返回帧错误码
}

10.3 浮点扩展 modbus_client.c

Modbus 规范不原生支持浮点数。Zephyr 通过约定实现： 浮点寄存器地址从 5000 开始（MODBUS_FP_EXTENSIONS_ADDR）， 每个浮点值占用两个 16 位寄存器（大端序 IEEE 754）。 

int modbus_read_holding_regs_fp(iface, unit_id, start_addr, reg_buf, num_regs)
{
    // 将 float 指针视为 uint16 指针
    uint16_t *reg_buf_u16 = (uint16_t *)reg_buf;

    err = modbus_read_holding_regs(iface, unit_id,
                                   start_addr + MODBUS_FP_EXTENSIONS_ADDR,
                                   reg_buf_u16, num_regs * 2);
    // 大端序已在 modbus_read_holding_regs 中转换
    // 浮点值的字节序由平台 float 内存表示决定
}

11服务器功能码处理

modbus_server_handler() 是服务器的核心分发函数。 它从接收到的 ADU 中提取功能码，分发到对应的处理函数。

11.1 分发逻辑 modbus_server.c

bool modbus_server_handler(struct modbus_context *ctx)
{
    // 1. 检查 CRC/帧错误
    if (ctx->rx_adu_err != 0) {
        LOG_WRN("Server has dropped frame");
        return false;
    }

    // 2. 检查 unit_id 匹配（0 = 广播，始终处理）
    if (ctx->rx_adu.unit_id != ctx->unit_id &&
        ctx->rx_adu.unit_id != 0)
        return false;

    // 3. 更新统计
    if (IS_ENABLED(CONFIG_MODBUS_FC08_DIAGNOSTIC))
        modbus_update_stats(ctx);

    // 4. 准备响应 ADU
    ctx->tx_adu.unit_id = ctx->rx_adu.unit_id;
    ctx->tx_adu.fc      = ctx->rx_adu.fc;

    // 5. 功能码分发
    switch (ctx->rx_adu.fc) {
    case MODBUS_FC01_COIL_RD:       exc = fc01_coil_read(ctx);       break;
    case MODBUS_FC02_DI_RD:         exc = fc02_di_read(ctx);         break;
    case MODBUS_FC03_HOLDING_REG_RD: exc = fc03_holding_reg_read(ctx); break;
    case MODBUS_FC04_IN_REG_RD:     exc = fc04_in_reg_read(ctx);     break;
    case MODBUS_FC05_COIL_WR:       exc = fc05_coil_write(ctx);      break;
    case MODBUS_FC06_HOLDING_REG_WR: exc = fc06_holding_reg_write(ctx); break;
    case MODBUS_FC08_DIAGNOSTICS:   exc = fc08_diagnostics(ctx);     break;
    case MODBUS_FC15_COILS_WR:      exc = fc15_coils_write(ctx);     break;
    case MODBUS_FC16_HOLDING_REGS_WR: exc = fc16_holding_regs_write(ctx); break;
    default:
        // 6. 尝试用户自定义功能码
        exc = modbus_user_defined_fc_handler(ctx);
        break;
    }

    // 7. 异常处理
    if (exc != MODBUS_EXC_NONE)
        modbus_excep_response(ctx, ctx->rx_adu.fc, exc);

    // 8. 广播地址不回复
    if (ctx->rx_adu.unit_id == 0)
        return false;

    return (exc == MODBUS_EXC_NONE);
}

14Raw ADU 模式与 TCP 网关

Raw ADU 模式提供不经过串行线路的原始 ADU 传输接口， 主要用于构建 Modbus TCP 网关。网关接收 TCP 连接上的 MBAP 帧， 通过 Raw 接口转发到 RTU 后端。

14.1 MBAP 头解析 modbus_raw.c:79-101

void modbus_raw_get_header(struct modbus_adu *adu, const uint8_t *header)
{
    adu->trans_id = sys_get_be16(&header[0]);   // 事务 ID
    adu->proto_id = sys_get_be16(&header[2]);   // 协议 ID (0x0000)
    adu->length = sys_get_be16(&header[4]);     // 后续长度
    adu->unit_id = header[6];                    // 从站地址
    adu->fc = header[7];                         // 功能码
    adu->length -= 2;  // 减去 unit_id + fc 的长度偏差
}

modbus_raw_backend_txn() 是 TCP 网关的核心函数， 将 TCP 侧的 ADU 通过 RTU 客户端接口发送到串行总线。 

int modbus_raw_backend_txn(const int iface, struct modbus_adu *adu)
{
    ctx = modbus_get_context(iface);

    // 仅支持配置为客户端的 RTU/ASCII 串行接口
    if (ctx->client == false ||
        (ctx->mode != MODBUS_MODE_RTU && ctx->mode != MODBUS_MODE_ASCII))
        return -ENOTSUP;

    // 将 ADU 复制到 RTU 接口的 TX 缓冲区
    memcpy(&ctx->tx_adu, adu, sizeof(struct modbus_adu));
    err = modbus_tx_wait_rx_adu(ctx);    // 发送并等待 RTU 响应

    if (err == 0) {
        // 保留 TCP 侧的 trans_id 和 proto_id
        uint16_t trans_id = adu->trans_id;
        uint16_t proto_id = adu->proto_id;
        memcpy(adu, &ctx->rx_adu, sizeof(struct modbus_adu));
        adu->trans_id = trans_id;
        adu->proto_id = proto_id;
    } else {
        // 目标设备无响应
        modbus_set_exception(adu, MODBUS_EXC_GW_TARGET_FAILED_TO_RESP);
    }
}

// 2. 初始化为客户端
const struct modbus_iface_param client_param = {
    .mode = MODBUS_MODE_RTU,
    .rx_timeout = 50000,        // 50ms 响应超时
    .serial = {
        .baud = 19200,
        .parity = UART_CFG_PARITY_NONE,
    },
};
modbus_init_client(client_iface, client_param);

// 3. 写保持寄存器 (FC16)
modbus_write_holding_regs(client_iface, node, 0, holding_reg, 8);

// 4. 读保持寄存器 (FC03)
modbus_read_holding_regs(client_iface, node, 0, holding_reg, 8);

// 5. 循环读写线圈 (FC01, FC05, FC15)
while (true) {
    modbus_read_coils(client_iface, node, 0, coil, coil_qty);
    modbus_write_coil(client_iface, node, addr, true);
    modbus_write_coils(client_iface, node, 0, coil, coil_qty);
    k_msleep(250);
}

16.2 RTU 服务器示例 samples/subsys/modbus/rtu_server/

演示服务器用户回调实现：

// 用户回调实现
static int coil_rd(uint16_t addr, bool *state) {
    *state = coils_state & BIT(addr);
    return 0;
}

static int coil_wr(uint16_t addr, bool state) {
    if (state) coils_state |= BIT(addr);
    else       coils_state &= ~BIT(addr);
    gpio_pin_set(led_dev[addr].port, led_dev[addr].pin, (int)state);
    return 0;
}

static int holding_reg_rd(uint16_t addr, uint16_t *reg) {
    *reg = holding_reg[addr];
    return 0;
}

static int holding_reg_wr(uint16_t addr, uint16_t reg) {
    holding_reg[addr] = reg;
    return 0;
}

// 注册回调并初始化
static struct modbus_user_callbacks mbs_cbs = {
    .coil_rd = coil_rd,
    .coil_wr = coil_wr,
    .holding_reg_rd = holding_reg_rd,
    .holding_reg_wr = holding_reg_wr,
};

const static struct modbus_iface_param server_param = {
    .mode = MODBUS_MODE_RTU,
    .server = {
        .user_cb = &mbs_cbs,
        .unit_id = 1,
    },
    .serial = {
        .baud = 19200,
        .parity = UART_CFG_PARITY_NONE,
    },
};

modbus_init_server(iface, server_param);

16.3 TCP 网关示例 samples/subsys/modbus/tcp_gateway/

展示如何使用 Raw ADU 接口构建 Modbus TCP → RTU 网关：

// 初始化 RTU 后端为客户端
backend = modbus_iface_get_by_name(bend_name);
modbus_init_client(backend, backend_param);

// TCP 服务器主循环
serv = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
bind(serv, ..., MODBUS_TCP_PORT);  // 端口 502
listen(serv, 5);

while (1) {
    client = accept(serv, ...);

    do {
        // 1. 接收 MBAP 头 (7 字节) + 功能码 (1 字节)
        recv(client, header, MODBUS_MBAP_AND_FC_LENGTH, MSG_WAITALL);
        modbus_raw_get_header(&tmp_adu, header);

        // 2. 接收数据域
        recv(client, tmp_adu.data, data_len, MSG_WAITALL);

        // 3. 通过 RTU 后端转发
        modbus_raw_backend_txn(backend, &tmp_adu);

        // 4. 回复 TCP 客户端
        modbus_raw_put_header(&tmp_adu, header);
        send(client, header, sizeof(header), 0);
        send(client, tmp_adu.data, tmp_adu.length, 0);
    } while (!rc);

    close(client);
}
TCP 网关架构 TCP 网关应用需要两个接口：
  • 前端接口：标准 BSD Socket 监听 TCP 502 端口
  • 后端接口：配置为 RTU 客户端的 Modbus 串行接口
modbus_raw_backend_txn() 是连接两者的桥梁。

17.4 关键设计特征

设计特征实现方式优势
零动态分配 设备树宏生成静态上下文表 无堆碎片风险，确定性内存
线程安全 k_mutex 保护客户端接口 多线程安全共享接口
模式隔离 RTU/ASCII/RAW 三模式独立函数集 编译期裁剪，无运行时开销
双 UART API IRQ + Async 编译期选择 适配不同硬件能力
回调驱动 服务器通过 user_callbacks 代理数据访问 应用层完全控制数据模型
TCP 网关就绪 Raw ADU + MBAP 头处理 零拷贝 TCP-RTU 桥接
可扩展功能码 sys_slist 运行时注册 无需修改子系统源码
