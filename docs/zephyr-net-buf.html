01设计动机与核心思想

在网络协议栈中，数据包在协议层之间传递时，每层可能添加或移除头部/尾部。 如果每次都复制整个数据包，性能开销巨大。Zephyr 的 net_buf 借鉴了 Linux sk_buff 的设计思想，但更轻量：

设计目标实现方式
零拷贝头部操作通过移动 data 指针而非移动数据实现 push/pull
安全共享引用计数，多个消费者共享同一 buffer
大包支持碎片链，多个小 buffer 链式组成大包
无碎片分配池化预分配，避免运行时 heap 碎片
协议元数据user_data 柔性数组，零成本携带上下文
灵活数据源支持外部数据、固定池、变长池、堆分配
net_buf 的使用者 Zephyr 中以下子系统大量使用 net_buf：
  • Bluetooth (HCI, ATT, L2CAP, SMP)
  • 网络协议栈 (IPv4/IPv6, TCP, UDP, CoAP)
  • USB 设备栈 (device_next)
  • CAN 总线
  • Modbus、COBS 等工具库

03net_buf_simple — 轻量缓冲区

net_buf_simple 是最基础的缓冲区表示，仅 16 字节（32 位系统）， 不含引用计数、碎片链、内存池关联。

3.1 结构体定义 net_buf.h:89

struct net_buf_simple {
    uint8_t *data;      // 有效数据起始指针（可前后移动）
    uint16_t len;       // 有效数据长度
    uint16_t size;      // 存储区总容量
    uint8_t *__buf;     // 存储区基地址（固定不变）
};

3.2 内存布局

  __buf                data                          data + len
    │                   │                               │
    ▼                   ▼                               ▼
    ├─── headroom ──────┤──────── data ─────────────────┤──── tailroom ────┤
    │                   │◄──── len ────►│               │                 │
    │                   │                               │                 │
    └───────────────────┴───────────────────────────────┴─────────────────┘
    ◄──────────────────────── size ──────────────────────────────────────►

  headroom = data - __buf                  （已预留/已 pull 的空间）
  tailroom = size - headroom - len          （可 add 的空间）
  max_len  = size - headroom               （当前 data 后最大可用长度）

3.3 关键特征

  • 无引用计数：生命周期由调用者管理
  • 无内存池：通常在栈上或静态分配
  • 无碎片链：单缓冲区
  • 与 net_buf ABI 兼容：net_buf 通过 union 包含 net_buf_simple

3.4 栈分配宏

// 在栈上定义一个 128 字节的 net_buf_simple
NET_BUF_SIMPLE_DEFINE(my_buf, 128);

// 带初始化的栈分配（可指定 reserve_head）
struct net_buf_simple *buf = NET_BUF_SIMPLE(64);
net_buf_simple_init(buf, 4);   // 预留 4 字节 headroom

struct net_buf {
    sys_snode_t   node;           // 链表节点（用于 FIFO/SList 传递）

            uint16_t  len;        // 有效数据长度
            uint16_t  size;       // 数据区容量
            uint8_t  *__buf;      // 数据区基地址

        struct net_buf_simple b;  // 作为 net_buf_simple 访问
    };

    uint8_t user_data[] __net_buf_align; // 柔性数组：协议元数据
};

05net_buf_pool — 内存池管理

5.1 结构体定义 net_buf.h:1088

struct net_buf_pool {
    struct k_lifo    free;              // 空闲 buffer 的 LIFO 队列
    struct k_spinlock lock;             // 并发保护自旋锁
    const uint16_t   buf_count;         // 池中 buffer 总数
    uint16_t         uninit_count;      // 未初始化 buffer 数（惰性初始化）
    uint8_t          user_data_size;    // 每个 buffer 的 user_data 大小
#if defined(CONFIG_NET_BUF_POOL_USAGE)
    atomic_t         avail_count;       // 当前可用数量
    const uint16_t   pool_size;         // 池总大小（调试用）
    uint16_t         max_used;          // 历史最大使用数
    const char      *name;              // 池名称（日志用）
#endif
    void (*const destroy)(struct net_buf *); // 可选销毁回调
    const struct net_buf_data_alloc *alloc;  // 数据区分配策略
    struct net_buf * const __bufs;           // net_buf 结构数组
};

5.2 惰性初始化

池创建时 uninit_count = buf_count，所有 buffer 尚未初始化。 首次分配时按需从 __bufs 数组中取出并初始化，避免启动时全部构造。 

// buf.c:72 — 惰性取 buffer
static inline struct net_buf *pool_get_uninit(struct net_buf_pool *pool,
                                              uint16_t uninit_count)
{
    size_t struct_size = ROUND_UP(sizeof(struct net_buf) + pool->user_data_size,
                                  __alignof__(struct net_buf));
    size_t byte_offset = (pool->buf_count - uninit_count) * struct_size;
    struct net_buf *buf = (struct net_buf *)(((uint8_t *)pool->__bufs) + byte_offset);

    buf->pool_id = pool_id(pool);
    buf->user_data_size = pool->user_data_size;
    return buf;
}
两次尝试优化 分配时，如果池中有未初始化 buffer 但 LIFO 中可能有之前归还的 buffer， 会先尝试 K_NO_WAIT 从 LIFO 取（更快），取不到再用未初始化 buffer。 这避免了不必要的初始化开销。 buf.c:276

06数据区分配策略 (FIXED / VAR / HEAP)

每个 buffer 的数据区（存放实际 payload 的区域）由独立的分配策略管理， 通过 net_buf_data_cb 和 net_buf_data_alloc 抽象。

6.1 回调接口 net_buf.h:1067

struct net_buf_data_cb {
    uint8_t* (*alloc)(struct net_buf *buf, size_t *size, k_timeout_t timeout);
    uint8_t* (*ref)(struct net_buf *buf, uint8_t *data);   // 可选
    void     (*unref)(struct net_buf *buf, uint8_t *data);
};

struct net_buf_data_alloc {
    const struct net_buf_data_cb *cb;
    void  *alloc_data;       // 池/堆指针
    size_t max_alloc_size;    // 最大分配大小（0 = 无限制）
    size_t alignment;         // 对齐要求（0 = 默认）
};

6.2 三种策略对比

特性 FIXED VAR HEAP
数据来源预分配字节数组k_heap 动态分配k_malloc 堆分配
数据大小固定（max_alloc_size）变长（≤池总量）变长（≤堆总量）
引用计数 (ref cb)不支持支持支持
阻塞等待不支持（K_NO_WAIT）支持不支持
内存碎片无堆内可能系统堆可能
典型用途BT HCI、USB网络协议栈通用

6.3 FIXED 策略实现 buf.c:165

// 直接按 buffer 索引计算数据区偏移，O(1) 分配
static uint8_t *fixed_data_alloc(struct net_buf *buf, size_t *size, k_timeout_t timeout)
{
    struct net_buf_pool *pool = net_buf_pool_get(buf->pool_id);
    const struct net_buf_pool_fixed *fixed = pool->alloc->alloc_data;

    *size = pool->alloc->max_alloc_size;
    return fixed->data_pool + *size * net_buf_id(buf);
}

6.4 VAR/HEAP 策略 — 引用计数隐藏在数据前

  VAR/HEAP 数据区内存布局:

  ┌─────────┬─────────────────────────────────────────────────┐
  │ref_count│              data payload                       │
  │ (1 字节) │              (size 字节)                        │
  └─────────┴─────────────────────────────────────────────────┘
  ▲         ▲
  │         │
  │         └── __buf = ref_count + GET_ALIGN(pool)
  │              data 指向此区域内的某个位置
  │
  └── 对齐到 sizeof(void*) 或 pool->alignment

  clone 时: data_ref() 递增 ref_count，共享同一数据区
  unref 时: ref_count--，到 0 时 k_heap_free / k_free

8.1 完整分配流程 buf.c:252

net_buf_alloc_fixed(pool, timeout)
  │
  └─ net_buf_alloc_len(pool, max_alloc_size, timeout)
       │
       ├─ 1. k_spin_lock(&pool->lock)
       │
       ├─ 2. 有未初始化 buffer？ (uninit_count > 0)
       │     ├─ YES:
       │     │    ├─ 非首次？→ 尝试 k_lifo_get(&pool->free, K_NO_WAIT)
       │     │    │    ├─ 成功 → 跳到 success
       │     │    │    └─ 失败 → uninit_count--, pool_get_uninit()
       │     │    └─ 跳到 success
       │     │
       │     └─ NO:
       │          k_spin_unlock
       │          buf = k_lifo_get(&pool->free, timeout)
       │          └─ 失败 → return NULL
       │
       ├─ 3. success:
       │     ├─ data_alloc(buf, &size, timeout)      # 调用池的 alloc 回调
       │     │    └─ 失败 → net_buf_destroy(buf), return NULL
       │     ├─ buf->ref = 1
       │     ├─ buf->frags = NULL
       │     ├─ buf->flags = 0
       │     ├─ memset(buf->user_data, 0, ...)
       │     ├─ buf->data = buf->__buf               # data 指向存储区起始
       │     └─ buf->len = 0
       │
       ├─ 4. [调试] 更新 avail_count, max_used
       │
       └─ return buf

8.2 释放流程 buf.c:440

net_buf_unref(buf)
  │
  ├─ assert(buf->ref > 0)       # 检测 double free
  │
  ├─ if (--buf->ref > 0)        # 引用计数仍 > 0
  │    └─ return                # 不释放
  │
  ├─ buf->data = NULL
  ├─ buf->frags = NULL
  │
  ├─ pool->destroy ?            # 有自定义销毁回调？
  │    ├─ pool->destroy(buf)    # 由回调负责调 net_buf_destroy()
  │    └─ net_buf_destroy(buf)  # 归还到 pool->free LIFO
  │
  └─ buf = frags                # 继续释放碎片链中的下一个 fragment!
       └─ 循环...
unref 自动释放碎片链 net_buf_unref() 会沿着 frags 链自动释放所有 fragment。 无需手动逐个释放碎片。但这也意味着如果某个 fragment 被 net_buf_ref() 增加了引用， 它的 ref > 0 不会被释放，只有 ref 降到 0 的 buffer 才归还池。

9.2 克隆 buf.c:494

struct net_buf *net_buf_clone(struct net_buf *buf, k_timeout_t timeout);

克隆行为取决于池的数据分配策略：

pool->alloc->cb->ref 是否存在？ (VAR / HEAP)
│
├─ YES (有 ref callback):
│    ├─ clone->__buf = data_ref(buf, buf->__buf)     # 数据区引用计数 +1
│    ├─ clone->data = buf->data                       # 共享同一数据区
│    ├─ clone->len = buf->len
│    ├─ clone->size = buf->size
│    └─ memcpy(clone->user_data, buf->user_data, ...) # 复制元数据
│
└─ NO (FIXED, 无 ref callback):
     ├─ clone->__buf = data_alloc(clone, &size, ...)  # 新分配数据区
     ├─ clone->data = clone->__buf + headroom(buf)     # 保持相同偏移
     ├─ net_buf_add_mem(clone, buf->data, buf->len)    # 完整数据拷贝
     └─ memcpy(clone->user_data, buf->user_data, ...)  # 复制元数据
所有权规则 net_buf_clone() 返回的新 buffer 与原 buffer 完全独立。 两者各自 unref()，互不影响。 对于 VAR/HEAP 池，数据区共享但引用计数保护了生命周期。

11碎片链 (Fragment Chain)

当数据包超过单个 buffer 容量时，使用碎片链。每个 net_buf 的 frags 指针指向下一个 fragment，形成单向链表。

11.1 链式结构

net_buf (head)                   net_buf (frag 1)              net_buf (frag 2)
┌──────────────────┐             ┌──────────────────┐          ┌──────────────────┐
│ node             │             │ node             │          │ node             │
│ frags ────────────────────────►│ frags ──────────────────────►│ frags = NULL    │
│ ref = 2          │             │ ref = 1          │          │ ref = 1          │
│ pool_id          │             │ pool_id          │          │ pool_id          │
│ ┌──────────────┐ │             │ ┌──────────────┐ │          │ ┌──────────────┐ │

11.3 自动碎片创建 — net_buf_append_bytes buf.c:666

// 追加超过单个 buffer 容量的数据，自动创建碎片
size_t net_buf_append_bytes(struct net_buf *buf, size_t len,
                            const void *value, k_timeout_t timeout,
                            net_buf_allocator_cb allocate_cb, void *user_data)
{
    struct net_buf *frag = net_buf_frag_last(buf);

    do {
        // 1. 填满当前 fragment 的 tailroom
        uint16_t count = min(len, net_buf_tailroom(frag));
        net_buf_add_mem(frag, value8, count);
        len -= count;

        if (len == 0) return added_len;

        // 2. 从同一池分配新 fragment
        frag = net_buf_alloc_len(pool, min(len, max_size), timeout);
        net_buf_frag_add(buf, frag);   // 挂到链尾
    } while (1);
}

11.4 碎片链线性化 — net_buf_linearize buf.c:626

// 将碎片链复制到连续内存
size_t net_buf_linearize(void *dst, size_t dst_len,
                         const struct net_buf *src,
                         size_t offset, size_t len)
{
    // 1. 跳过 offset 字节找到起始 fragment
    while (frag && offset >= frag->len) {
        offset -= frag->len;
        frag = frag->frags;
    }

    // 2. 逐 fragment 拷贝
    while (frag && len > 0) {
        to_copy = min(len, frag->len - offset);
        memcpy(dst + copied, frag->data + offset, to_copy);
        copied += to_copy;
        len -= to_copy;
        frag = frag->frags;
        offset = 0;
    }
    return copied;
}

12user_data 元数据

每个 buffer 尾部的 user_data[] 柔性数组用于存放协议相关的元数据。 大小在池定义时确定，分配时自动清零。 

// 定义自定义 user_data
struct bt_l2cap_chan;   // 前向声明

struct bt_acl_ud {
    struct bt_l2cap_chan *chan;   // L2CAP 通道
    uint8_t  att_type;           // ATT 操作类型
    uint16_t acl_id;             // ACL 连接 ID
};

// 创建池时指定 user_data 大小
NET_BUF_POOL_DEFINE(acl_pool, 8, 256, sizeof(struct bt_acl_ud), NULL);

// 分配后访问
struct net_buf *buf = net_buf_alloc_fixed(&acl_pool, K_FOREVER);
struct bt_acl_ud *ud = net_buf_user_data(buf);
ud->chan = my_chan;
ud->att_type = BT_ATT_WRITE_CMD;

// net_buf_user_data() 是一个简单的宏：
#define net_buf_user_data(buf) ((void *)(buf)->user_data)
内存对齐 user_data[] 带有 __net_buf_align 属性 （默认 __aligned(sizeof(void*))），确保结构体指针正确对齐。 定义池时用 BUILD_ASSERT 验证偏移和大小。

14线程安全与同步

保护对象同步原语说明
池操作（分配/释放） k_spinlock 保护 uninit_count 和 free LIFO
池 LIFO 队列 k_lifo（内部自旋锁） 支持阻塞等待和超时
sys_slist 操作 net_buf_slist_lock 全局自旋锁保护 net_buf_slist_put/get
引用计数 原子性保证 单线程内递增/递减，由调用者保证可见性
可用计数 atomic_t avail_count 使用原子操作
buffer 数据操作的线程安全 net_buf_add/push/pull 等不是线程安全的。 同一 buffer 在多线程间共享时，调用者需自行加锁。 但通过 FIFO 传递 buffer（net_buf_put/get）是线程安全的。

15.1 定义池与基础分配

#include <zephyr/net_buf.h>

/* 1. 定义协议头部 */
struct proto_hdr {
    uint8_t  type;
    uint8_t  seq;
    uint16_t length;
} __packed;

/* 2. 定义 user_data（存放协议上下文）*/
struct my_ud {
    struct proto_hdr hdr;
    uint8_t  priority;
    uint16_t src_port;
};

/* 3. 定义固定大小池：8 个 buffer，每个 256 字节数据区 */
NET_BUF_POOL_DEFINE(my_pool, 8, 256, sizeof(struct my_ud), NULL);

void example_basic(void)
{
    /* 4. 分配 buffer */
    struct net_buf *buf = net_buf_alloc_fixed(&my_pool, K_FOREVER);

    /* 5. 设置 user_data */
    struct my_ud *ud = net_buf_user_data(buf);
    ud->hdr.type = 0x01;
    ud->hdr.seq = 42;
    ud->priority = 3;

    /* 6. 追加 payload */
    uint8_t payload[] = {0xDE, 0xAD, 0xBE, 0xEF};
    net_buf_add_mem(buf, payload, sizeof(payload));
    // buf->len = 4

    /* 7. 释放 */
    net_buf_unref(buf);
}

15.2 零拷贝协议封装（利用 headroom）

void example_protocol_encap(void)
{
    struct net_buf *buf = net_buf_alloc_fixed(&my_pool, K_FOREVER);

    /* Step 1: 先写 payload（填满尾部）*/
    net_buf_add_mem(buf, app_data, app_data_len);

    /* Step 2: 传输层头部 — 利用 headroom，零拷贝前插 */
    net_buf_push_be16(buf, app_data_len);   // length 字段
    net_buf_push_u8(buf, 0x01);             // type 字段

    /* Step 3: 网络层头部 — 继续利用 headroom */
    net_buf_push_be16(buf, 0x0800);         // EtherType
    net_buf_push_mem(buf, dst_mac, 6);      // 目标 MAC
    net_buf_push_mem(buf, src_mac, 6);      // 源 MAC

    /* 此时 buf->data 指向 Ethernet 头部
       buf->len = 14 + 3 + app_data_len */

    send_packet(buf);
    net_buf_unref(buf);
}

15.4 分片处理大数据

void example_fragments(void)
{
    struct net_buf *head = net_buf_alloc_fixed(&my_pool, K_FOREVER);

    /* 追加 1024 字节 — 超出单个 buffer 容量，自动创建碎片 */
    uint8_t big_data[1024];
    memset(big_data, 0xAA, sizeof(big_data));

    net_buf_append_bytes(head, sizeof(big_data), big_data,
                         K_FOREVER, NULL, NULL);

    /* 线性化到连续内存 */
    uint8_t linear[1024];
    size_t copied = net_buf_linearize(linear, sizeof(linear),
                                      head, 0, sizeof(linear));
    // copied = 1024

    /* unref head 会自动释放整个碎片链 */
    net_buf_unref(head);
}

15.5 线程间传递

/* 定义 FIFO */
K_FIFO_DEFINE(rx_fifo);

/* 生产者线程 */
void producer_thread(void)
{
    while (1) {
        struct net_buf *buf = net_buf_alloc_fixed(&my_pool, K_FOREVER);
        /* ... 填充数据 ... */
        net_buf_put(&rx_fifo, buf);    // 转移所有权，线程安全
    }
}

/* 消费者线程 */
void consumer_thread(void)
{
    while (1) {
        struct net_buf *buf = net_buf_get(&rx_fifo, K_FOREVER);
        /* ... 处理数据 ... */
        net_buf_unref(buf);            // 用完释放 */
    }
}

15.6 引用计数共享

void example_refcount(void)
{
    struct net_buf *buf = net_buf_alloc_fixed(&my_pool, K_FOREVER);
    net_buf_add_mem(buf, shared_data, len);

    // 模块 A 和 B 都需要处理这个数据包
    struct net_buf *buf_a = buf;           // 原始引用
    struct net_buf *buf_b = net_buf_ref(buf); // 增加引用，ref = 2

    // 模块 A 处理完毕
    net_buf_unref(buf_a);   // ref: 2 → 1，不释放

    // 模块 B 处理完毕
    net_buf_unref(buf_b);   // ref: 1 → 0，真正释放回池
}

16.3 设计特征总结

设计特征实现方式优势
零拷贝头部操作 data 指针移动（push 左移 / pull 右移） 协议层封装/解封装无需内存拷贝
预分配池化 编译期静态分配 net_buf 数组 + 数据数组 无运行时碎片，确定性延迟
惰性初始化 uninit_count 递减，首次使用才构造 加速启动，避免 memset 全池
引用计数 uint8_t ref，net_buf_ref/unref 多模块安全共享同一 buffer
碎片链 单向链表 frags 指针 支持超过单个 buffer 的大数据包
灵活数据源 FIXED / VAR / HEAP / 外部数据 适配不同场景的内存需求
协议元数据 user_data[] 柔性数组 零额外指针开销携带上下文
线程安全传递 k_lifo / k_fifo + 自旋锁 多线程生产者-消费者模式
