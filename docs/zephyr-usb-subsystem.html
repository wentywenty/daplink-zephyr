02核心线程模型 usbd_core.c

usbd_core.c 是整个 USB 设备栈的事件调度中枢。它在系统启动时通过 SYS_INIT 创建一个协作式线程，以消息队列驱动的事件循环方式处理所有 USB 事件。

线程创建与初始化

在 POST_KERNEL 阶段，usbd_pre_init() 函数被调用，完成以下关键工作：

/* usbd_core.c:284-307 */
static int usbd_pre_init(void)
{
    k_thread_create(&usbd_thread_data, usbd_stack,
            K_KERNEL_STACK_SIZEOF(usbd_stack),
            usbd_thread,
            NULL, NULL, NULL,

        atomic_set(&c_nd->state, 0);
    }
    STRUCT_SECTION_FOREACH_ALTERNATE(usbd_class_hs, usbd_class_node, c_nd) {
        atomic_set(&c_nd->state, 0);
    }
    return 0;

协作式线程优先级 K_PRIO_COOP(8)

USB 设备栈使用协作式线程（优先级数字越小越高），优先级 8 确保它高于大多数应用线程但低于关键的系统和中断处理。线程不会在处理事件时被抢占，保证了 USB 事件处理的原子性。

事件队列定义

消息队列 usbd_msgq 是 UDC 驱动与 USB 设备栈之间的唯一通信通道。队列中每个元素是一个 struct udc_event，最大容量由 CONFIG_USBD_MAX_UDC_MSG 决定。

事件载体回调

usbd_event_carrier() 是注册到 UDC 驱动层的回调函数。每当 UDC 驱动产生事件（ISR 上下文），该函数被调用：

/* usbd_core.c:34-51 */
static int usbd_event_carrier(const struct device *dev,
                const struct udc_event *const event)
{
    struct usbd_context *const uds_ctx = (void *)udc_get_event_ctx(dev);
    k_spinlock_key_t key;

    if (event->type == UDC_EVT_EP_REQUEST) {
        /* 将已完成的传输请求添加到 slist，
         * 以确保它们不会丢失 */
        key = k_spin_lock(&uds_ctx->ep_event_lock);
        sys_slist_append(&uds_ctx->ep_events, &event->buf->node);
        k_spin_unlock(&uds_ctx->ep_event_lock, key);
    }

    return k_msgq_put(&usbd_msgq, event, K_NO_WAIT);
}
双队列设计

对于 EP_REQUEST 事件（传输完成），数据缓冲区除了放入主消息队列外，还额外加入 ep_events 单链表。这个 slist 由自旋锁 ep_event_lock 保护，确保 ISR 上下文下的安全操作。这一设计允许在处理其他事件类型时也能检查是否有待处理的传输完成事件。

事件处理主循环

/* usbd_core.c:209-226 */
static void usbd_thread(void *p1, void *p2, void *p3)
{
    struct usbd_context *uds_ctx;
    struct udc_event event;

    while (true) {
        k_msgq_get(&usbd_msgq, &event, K_FOREVER);

        uds_ctx = (void *)udc_get_event_ctx(event.dev);
        __ASSERT(uds_ctx != NULL && usbd_is_initialized(uds_ctx),
             "USB device is not initialized");
        usbd_event_handler(uds_ctx, &event);
    }
}

事件分发流程

usbd_event_handler() 是事件分发中心（usbd_core.c:155-207），其处理逻辑：

  1. 首先调用 event_handler_ep_request() 检查 ep_events 链表，处理所有已完成的传输请求
  2. 如果当前事件本身就是 EP_REQUEST，则直接返回（已在步骤 1 处理）
  3. 根据事件类型分发：
    • VBUS_READY/REMOVED → 发布简单消息给应用回调
    • SUSPEND/RESUME → 更新挂起状态 → 广播给所有类 → 发布消息
    • SOF → 广播给所有类（用于 HID 空闲速率等定时功能）
    • RESET → 总线重置处理 → 发布消息
    • ERROR → 发布错误消息

03设备生命周期状态机 usbd_device.c

usbd_device.c（482 行）管理 USB 设备的完整生命周期，提供四个核心 API：usbd_init()、usbd_enable()、usbd_disable()、usbd_shutdown()。

usbd_init() — 初始化

/* usbd_device.c:228-269 */
int usbd_init(struct usbd_context *const uds_ctx)
{
    int ret;

    /* 锁定调度器，确保上下文在完全初始化前不被抢占 */
    k_sched_lock();
    usbd_device_lock(uds_ctx);  // mutex lock

    if (uds_ctx->dev == NULL) {
        ret = -ENODEV;
        goto init_exit;
    }

    if (usbd_is_initialized(uds_ctx)) {
        ret = -EALREADY;
        goto init_exit;
    }

    if (!device_is_ready(uds_ctx->dev)) {
        ret = -ENODEV;
        goto init_exit;
    }

    ret = usbd_device_init_core(uds_ctx);  // udc_init + usbd_init_configurations

    memset(&uds_ctx->ch9_data, 0, sizeof(struct usbd_ch9_data));
    uds_ctx->status.initialized = true;

init_exit:
    usbd_device_unlock(uds_ctx);
    k_sched_unlock();
    return ret;
}

usbd_device_init_core()（usbd_core.c:228-247）执行两个关键操作：

  1. 调用 udc_init() 初始化底层控制器驱动，注册 usbd_event_carrier 回调
  2. 调用 usbd_init_configurations() 为所有配置分配端点地址和接口号

usbd_enable() — 使能设备

/* usbd_device.c:302-355 关键路径 */
int usbd_enable(struct usbd_context *const uds_ctx)
{
    /* 检查 EP0 队列是否为空（disable/enable 循环中复用） */
    ep0_empty = udc_ep_queue_is_empty(uds_ctx->dev, USB_CONTROL_EP_OUT);

    ret = udc_enable(uds_ctx->dev);

    /* 预分配 SETUP 缓冲区 */

    /* 初始化控制管道（仅当 EP0 为空时才入队 SETUP buffer） */
    ret = usbd_init_control_pipe(uds_ctx, ep0_empty);

    uds_ctx->status.enabled = true;
}

SETUP 缓冲区预分配策略

usbd_preallocate()（usbd_device.c:271-291）是关键的设计决策：

/* usbd_device.c:271-291 */
static int usbd_preallocate(struct usbd_context *const uds_ctx)
{
    /* SETUP buffer 分配不能失败。USB 栈在每次控制传输后
     * 有效重用同一个缓冲区。这确保了 SETUP 阶段始终可以
     * 接收，无论 UDC 内存使用情况如何。
     *
     * 预分配发生在 usbd_enable() 中，因为 bMaxPacketSize0
     * 必须在分配时已知。缓冲区保持到 usbd_shutdown()，
     * 不会在 udc_disable()/udc_enable() 循环中重新分配。
     */
    if (uds_ctx->setup_buf == NULL) {
        uds_ctx->setup_buf = udc_ctrl_setup_alloc(uds_ctx->dev);
    }
    if (uds_ctx->setup_buf == NULL) {
        return -ENOMEM;
    }
    return 0;
}
SETUP 缓冲区生命周期

缓冲区在 usbd_enable() 中分配，在 usbd_shutdown() 中释放。中间的 usbd_disable()/usbd_enable() 循环不会释放它。每次控制传输完成后，通过 net_buf_ref() + net_buf_reset() 复用同一缓冲区。

usbd_disable() / usbd_shutdown()

usbd_disable()：将配置重置为 0，禁用 UDC，设置 enabled=false。

usbd_shutdown()：调用 usbd_device_shutdown_core() 移除所有类、描述符、vendor request；调用 udc_purge_queues() 清空端点队列；释放预分配的 SETUP 缓冲区；设置 initialized=false。

04USB Chapter 9 协议引擎 usbd_ch9.c 1306 行

usbd_ch9.c 是整个 USB 设备栈中最大的文件，实现了完整的 USB 2.0 Chapter 9 协议。它管理控制传输的完整生命周期：SETUP → DATA IN/OUT → STATUS IN/OUT。

控制传输核心函数

int usbd_handle_ctrl_xfer(struct usbd_context *const uds_ctx,
              struct net_buf *const buf, int err)
{
    struct udc_buf_info *bi = udc_get_buf_info(buf);

    /* 1. 错误处理：释放 buffer，重新入队 SETUP */
    if (err) {
        net_buf_unref(buf);
        if (bi->setup || (bi->data && bi->ep == USB_CONTROL_EP_OUT))
            return usbd_enqueue_setup(uds_ctx);
        return 0;
    }

    /* 2. Data IN 完成：释放 buffer */
    if (bi->data && bi->ep == USB_CONTROL_EP_IN) {
        net_buf_unref(buf);
        return 0;
    }

    /* 3. SETUP / Data OUT 处理 */
    if ((bi->setup || bi->data) && bi->ep == USB_CONTROL_EP_OUT) {
        if (bi->setup) {
            ctrl_xfer_get_setup(uds_ctx, buf); // 解析 SETUP 包
            // ... 根据方向分配 Data IN/OUT buffer
        }

        ret = handle_setup_request(uds_ctx, next_buf); // 分发请求

        if (errno) goto ctrl_xfer_stall; // 协议错误 → STALL

        // 成功：入队响应（Status IN 或 Data IN）
        if (setup->wLength == 0)
            ret = usbd_enqueue_status_in(uds_ctx);
        else if (reqtype_is_to_device(setup))
            ret = usbd_enqueue_status_in(uds_ctx);
        else
            // Data IN + Status OUT
            usbd_enqueue_status_out(uds_ctx);

        return usbd_enqueue_setup(uds_ctx); // 重新入队 SETUP buffer
    }

    /* 4. Status OUT 完成 */
    if (bi->status && bi->ep == USB_CONTROL_EP_OUT) { ... }

    /* 5. Status IN 完成：执行 post_status 操作 */
    if (bi->status && bi->ep == USB_CONTROL_EP_IN) {
        if (uds_ctx->ch9_data.post_status)
            ret = post_status_stage(uds_ctx); // SET_ADDRESS / TEST_MODE
    }

ctrl_xfer_stall:
    // STALL 对应方向的端点，重新入队 SETUP
    udc_ep_set_halt(...);
    return usbd_enqueue_setup(uds_ctx);
}

SETUP 包解析

ctrl_xfer_get_setup()（usbd_ch9.c:1043-1059）提取并字节交换 SETUP 包的字段：

static int ctrl_xfer_get_setup(struct usbd_context *const uds_ctx,
                  struct net_buf *const buf)
{
    struct usb_setup_packet *setup = usbd_get_setup_pkt(uds_ctx);

    if (buf->len != sizeof(struct usb_setup_packet))
        return -EINVAL;

    memcpy(setup, buf->data, sizeof(struct usb_setup_packet));

    // 字节交换为 CPU 字节序
    setup->wValue  = sys_le16_to_cpu(setup->wValue);
    setup->wIndex  = sys_le16_to_cpu(setup->wIndex);
    setup->wLength = sys_le16_to_cpu(setup->wLength);

    return 0;
}

static int assign_ep_addr(const struct device *dev,
              struct usb_ep_descriptor *const ed,
              uint32_t *const config_ep_bm,
              uint32_t *const class_ep_bm)
{
    for (unsigned int idx = 1; idx < 16U; idx++) {
        uint8_t ep = USB_EP_DIR_IS_IN(ed->bEndpointAddress) ?
                     USB_EP_DIR_IN | idx : idx;

        // 跳过已分配的端点
        if (usbd_ep_bm_is_set(config_ep_bm, ep) ||
            usbd_ep_bm_is_set(class_ep_bm, ep))
            continue;

        // 尝试让 UDC 驱动验证此端点配置
        ret = udc_ep_try_config(dev, ep, ed->bmAttributes, &mps, ed->bInterval);

        if (ret == 0) {
            ed->bEndpointAddress = ep;  // 更新描述符中的端点地址
            ed->wMaxPacketSize = sys_cpu_to_le16(mps);
            usbd_ep_bm_set(class_ep_bm, ep);
            usbd_ep_bm_set(config_ep_bm, ep);
            return 0;
        }
    }
    return -ENODEV;
}

static void msg_work_handler(struct k_work *work)
{
    /* 1. 在 spinlock 保护下 peek 链表头部 */
    key = k_spin_lock(&ml_lock);
    node = sys_slist_peek_head(&msg_list);
    k_spin_unlock(&ml_lock, key);

    /* 2. 检查设备是否已初始化 */
    if (!usbd_is_initialized(m_pkt->ctx)) {
        /* 未初始化则延迟重试 */
        k_work_reschedule(dwork, K_MSEC(CONFIG_USBD_MSG_WORK_DELAY));
        return;
    }

    /* 3. 取出并执行回调 */
    key = k_spin_lock(&ml_lock);
    node = sys_slist_get(&msg_list);
    k_spin_unlock(&ml_lock, key);

    m_pkt->ctx->msg_cb(m_pkt->ctx, &m_pkt->msg);
    k_mem_slab_free(&usbd_msg_slab, (void *)m_pkt);

    /* 4. 如果还有消息，继续调度 */
    if (!sys_slist_is_empty(&msg_list))
        k_work_schedule(dwork, K_NO_WAIT);
}

5. 零拷贝缓冲管理

USB 栈使用 net_buf 链进行数据传输，通过 bi->owner 追踪缓冲区归属，实现零拷贝的类驱动分发：

  1. 类驱动调用 usbd_ep_enqueue() 时设置 bi->owner = c_data
  2. 传输完成后，usbd_class_handle_xfer() 通过 bi->owner 找到正确的类实例
  3. 调用 usbd_class_request(bi->owner, buf, err) 直接传递原始 buffer
  4. 类驱动处理完后调用 usbd_ep_buf_free() 释放
Owner Tracking 的优势

相比传统的"按端点号查找类"的方式，owner tracking 直接通过指针定位类实例，避免了线性搜索。这对于有多个类实例共享相似端点配置的复合设备尤为重要。

6. 可迭代段 (Iterable Sections) — 静态注册

USB 类通过链接器可迭代段实现编译时注册、运行时遍历的零开销模式：

// 编译时：USBD_DEFINE_CLASS 宏将类节点放入特殊段
USBD_DEFINE_CLASS(hid_0, &usbd_hid_api, priv, NULL);
    → __attribute__((section("usbd_class_fs")))

// 运行时：遍历段中的所有节点
STRUCT_SECTION_FOREACH_ALTERNATE(usbd_class_fs, usbd_class_node, c_nd) {
    // c_nd 自动指向段中的下一个节点
}

这种设计使得添加新的 USB 类无需修改核心代码，只需定义自己的类实例即可被自动发现和注册。
